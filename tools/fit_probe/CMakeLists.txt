# fit_probe: the surface fit traced (fit_probe.cpp). Desktop: -DOpenCV_DIR=<OpenCV 4.11 build>.
# Android (NDK toolchain): -DFS_OPENCV_ANDROID=<the org.opencv:opencv AAR's prefab/modules/opencv_java4>.
cmake_minimum_required(VERSION 3.18)
project(fit_probe CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CORE ${CMAKE_CURRENT_LIST_DIR}/../../core/src CACHE PATH "core sources (another tree's core/src to compare versions)")
file(GLOB SRC ${CORE}/*.cpp)
list(REMOVE_ITEM SRC ${CORE}/gpu.cpp ${CORE}/surface.cpp)
add_executable(fit_probe fit_probe.cpp ${SRC} ${CMAKE_CURRENT_LIST_DIR}/../desktop_check/gpu_stub.cpp)
target_include_directories(fit_probe PRIVATE ${CORE})
if(ANDROID)
  add_library(opencv_java4 SHARED IMPORTED)
  set_target_properties(opencv_java4 PROPERTIES IMPORTED_LOCATION ${FS_OPENCV_ANDROID}/libs/android.${ANDROID_ABI}/libopencv_java4.so
                        INTERFACE_INCLUDE_DIRECTORIES ${FS_OPENCV_ANDROID}/include)
  target_link_libraries(fit_probe PRIVATE opencv_java4 log)
else()
  find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d photo)
  target_link_libraries(fit_probe PRIVATE ${OpenCV_LIBS})
endif()
if(MSVC)
  target_compile_options(fit_probe PRIVATE /fp:precise /utf-8 /EHsc /bigobj)
  target_compile_definitions(fit_probe PRIVATE _USE_MATH_DEFINES NOMINMAX)
else()
  target_compile_options(fit_probe PRIVATE -ffp-contract=off)
endif()
add_executable(libm_probe libm_probe.cpp)
if(NOT MSVC)
  target_compile_options(libm_probe PRIVATE -ffp-contract=off)
endif()
add_executable(hyp_test hyp_test.cpp)
add_executable(exp_test exp_test.cpp)
add_executable(gb_test gb_test.cpp)
if(ANDROID)
  target_link_libraries(gb_test PRIVATE opencv_java4)
else()
  target_link_libraries(gb_test PRIVATE ${OpenCV_LIBS})
endif()
if(NOT MSVC)
  target_compile_options(gb_test PRIVATE -ffp-contract=off)
endif()
